    6. تدويل العروض

    سنتناول هنا مشكلة تدويل العروض. إنها مشكلة معقدة يمكن العثور على وصف جيد لها في المقالة التالية لـ Scott Hanselman:

    [http://www.hanselman.com/blog/GlobalizationInternationalizationAndLocalizationInASPNETMVC3JavaScriptAndJQueryPart1.aspx]

    لنبدأ أولاً بتعريفه للمصطلحات المختلفة المتعلقة بتدويل العروض:

    التدويل (i18n)
    جعل التطبيق يدعم لغات وإعدادات محلية مختلفة
    التوطين (l10n)
    جعل التطبيق يدعم لغة وإعدادات محلية محددة
    العولمة
    الجمع بين Internationalisation و Localisation
    اللغة
    اللغة المستخدمة – يُشار إليها برمز ISO (fr: الفرنسية، es: الإسبانية، en: الإنجليزية، ...)
    الإعدادات المحلية
    نوع من اللغة – يُشار إليه أيضًا برمز ISO (en_GB: الإنجليزية البريطانية، en_US: الإنجليزية الأمريكية، ...)

    6.2. إدارة الثقافة

    يتم التحكم في لغة العروض بواسطة الكائن [Thread.CurrentThread.CurrentUICulture]. لعرض الصفحات باللغة [fr-FR]، نكتب:

    Thread.CurrentThread.CurrentUICulture=new CultureInfo("fr-FR");

    يتم التحكم في الترجمة (التواريخ والأرقام والعملات والساعات، ...) بواسطة الكائن [Thread.CurrentThread.CurrentCulture]. وبطريقة مشابهة لما تمت كتابته سابقًا، نكتب:

    Thread.CurrentThread.CurrentCulture=new CultureInfo("fr-FR");

    يمكن أن تكون هاتان التعليمتان في منشئ كل وحدة تحكم في التطبيق. ولكن قد نرغب أيضًا في تحليل هذا الرمز المشترك بين جميع وحدات التحكم. سنتبع هذا المسار.

    نقوم بإنشاء عنصرين تحكم جديدين:

      
    • [I18NController] ستكون الفئة الأم لجميع وحدات التحكم التي تستخدم الترجمة؛
    • [SecondController] هي وحدة تحكم نموذجية مشتقة من [I18NController].

    رمز وحدة التحكم [I18NController] هو التالي:

    
using System.Threading;
using System.Web;
using System.Web.Mvc;

namespace Exemples.Controllers
{
  public abstract class I18NController : Controller
  {
    public I18NController()
    {
      // يتم استرداد سياق الطلب الحالي
      HttpContext httpContext = HttpContext.Current;
      // يتم فحص الطلب بحثًا عن المعلمة [lang]
      // البحث عنه في معلمات URL
      string langue = httpContext.Request.QueryString["lang"];
      if (langue == null)
      {
        // يتم البحث عنه في المعلمات المرسلة
        langue = httpContext.Request.Form["lang"];
      }
      if (langue == null)
      {
        // يتم البحث عنه في جلسة عمل المستخدم
        langue = httpContext.Session["lang"] as string;
      }
      if (langue == null)
      {
        // المعلمة الأولى من رأس الصفحة HTTP AcceptLanguages
        langue = httpContext.Request.UserLanguages[0];
      }
      if (langue == null)
      {
        // الثقافة fr-FR
        langue = "fr-FR";
      }
      // يتم تعيين اللغة في الجلسة
      httpContext.Session["lang"] = langue;
      // يتم تعديل الثقافات الخاصة بالخيط            
      Thread.CurrentThread.CurrentCulture = new System.Globalization.CultureInfo(langue);
      Thread.CurrentThread.CurrentUICulture = Thread.CurrentThread.CurrentCulture;
    }
  }
}
